cmake_minimum_required(VERSION 3.20)
project(mosaico_ui_sim LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
get_filename_component(APP_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(LV_BUILD_CONF_PATH "${CMAKE_CURRENT_LIST_DIR}/lv_conf.h" CACHE PATH "" FORCE)
set(CONFIG_LV_BUILD_DEMOS OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(CONFIG_LV_USE_THORVG_INTERNAL OFF CACHE BOOL "" FORCE)
find_package(PkgConfig REQUIRED)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
add_subdirectory("${APP_ROOT}/managed_components/lvgl__lvgl" lvgl)
target_link_libraries(lvgl PUBLIC PkgConfig::SDL2 m)
set(BOARD "${APP_ROOT}/main/boards/espressif/esp-mosaico")
add_executable(mosaico_ui_sim main.cc "${BOARD}/mosaico_ui.cc"
    "${BOARD}/mosaico_companion_asset.cc" "${BOARD}/mosaico_ui_small.c")
target_include_directories(mosaico_ui_sim PRIVATE "${BOARD}")
target_link_libraries(mosaico_ui_sim PRIVATE lvgl)
target_compile_options(mosaico_ui_sim PRIVATE -Wall -Wextra -Werror -UNDEBUG)
